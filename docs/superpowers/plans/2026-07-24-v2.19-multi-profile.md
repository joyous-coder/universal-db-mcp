# v2.19 Multi-Profile Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add SQLCipher-based profile encryption + cross-profile filtering for templates and history, without breaking any v2.14-v2.18 API.

**Architecture:** New `EncryptedSqliteBackend` abstraction in `src/utils/encrypted-sqlite.ts` exposes `NativeSqliteBackend` (reuses v2.16 multi-backend) and `CipherSqliteBackend` (uses `better-sqlite3-multiple-ciphers` with `PRAGMA key`). `ProfileStore` selects backend based on optional `cipherKey`. Templates and history stores get `profile_name` columns (nullable) with new filter options + `group_by='profile'` aggregation. `QueryAnalyzer` gains `setProfileProvider` to inject active profile into history records.

**Tech Stack:** TypeScript (ESM), fastify, MCP SDK, vitest, Node 22.5+ `node:sqlite` / `better-sqlite3` (already supported), optional `better-sqlite3-multiple-ciphers` (SQLCipher build).

## Global Constraints

- Backward compatible: existing v2.18 profiles.db (plaintext) still loads when `DB_PROFILE_ENCRYPTION_KEY` is unset.

```typescript
// src/utils/encrypted-sqlite.ts
import type { SQLiteConnection } from '../adapters/sqlite/types.js';
import { detectSqliteBackend as detectV16Backend } from '../adapters/sqlite/types.js';

export interface EncryptedSqliteBackend {
  name: 'native' | 'cipher';
  open(dbPath: string, options: { readonly?: boolean; cipherKey?: string }): Promise<SQLiteConnection>;
}

export class NativeSqliteBackend implements EncryptedSqliteBackend {
  name = 'native' as const;
  async open(dbPath: string, options: { readonly?: boolean }): Promise<SQLiteConnection> {
    const backend = await detectV16Backend();
    return backend.open(dbPath, { readonly: options.readonly ?? false });
  }
}

export class CipherSqliteBackend implements EncryptedSqliteBackend {
  name = 'cipher' as const;
  async open(dbPath: string, options: { readonly?: boolean; cipherKey?: string }): Promise<SQLiteConnection> {
    if (!options.cipherKey) throw new Error('CipherSqliteBackend requires cipherKey');
    let Database: any;
    try {
      const mod = await import('better-sqlite3-multiple-ciphers');
      Database = (mod as any).default ?? mod;
    } catch (err) {
      throw new Error('better-sqlite3-multiple-ciphers not installed; set DB_PROFILE_ENCRYPTION_KEY="" to use plaintext, or install the optional dependency');
    }
    const db = new Database(dbPath, { readonly: options.readonly ?? false });
    db.pragma(`cipher='sqlcipher'`);
    db.pragma(`key="${options.cipherKey.replace(/"/g, '\\"')}"`);
    // Verify key by querying cipher_version
    try {
      db.pragma('cipher_version');
    } catch (err) {
      db.close();
      throw new Error(`failed to decrypt profiles.db — check DB_PROFILE_ENCRYPTION_KEY`);
    }
    // Wrap in SQLiteConnection interface (reuse v2.16 wrapper pattern)
    return wrapBetterSqlite3(db);
  }
}

function wrapBetterSqlite3(db: any): SQLiteConnection {
  return {
    exec(sql) { db.exec(sql); },
    prepare(sql) {
      const stmt = db.prepare(sql);
      return {
        all(...params) { return stmt.all(...params) as Record<string, unknown>[]; },
        get(...params) { return stmt.get(...params) as Record<string, unknown> | undefined; },
        run(...params) {
          const r = stmt.run(...params);
          return { changes: Number(r.changes ?? 0), lastInsertRowid: Number(r.lastInsertRowid ?? 0) };
        },
      };
    },
    pragma(key, value) { db.pragma(`${key} = ${value}`); },
    close() { try { db.close(); } catch { /* ignore */ } },
  };
}

export function detectEncryptedBackend(cipherKey: string | undefined): EncryptedSqliteBackend {
  if (cipherKey && cipherKey.length > 0) {
    return new CipherSqliteBackend();
  }
  return new NativeSqliteBackend();
}
```

Note: This implementation will work IF `better-sqlite3-multiple-ciphers` is installed. Without it, `CipherSqliteBackend.open` throws a clear error. Tests for the cipher path will be skipped or guarded with `it.skipIf(!hasCip)` pattern if needed.

```typescript
import { detectEncryptedBackend } from '../utils/encrypted-sqlite.js';

export class ProfileStore {
  private conn: SQLiteConnection | null = null;
  private initPromise: Promise<void> | null = null;
  private cipherKey?: string;

  constructor(public readonly dbPath: string, options?: { cipherKey?: string }) {
    this.cipherKey = options?.cipherKey;
    if (this.cipherKey && this.cipherKey.length > 0 && this.cipherKey.length < 8) {
      console.warn(`[profile] WARNING: DB_PROFILE_ENCRYPTION_KEY is short (<8 chars). Use a stronger key.`);
    }
  }

  private async init(): Promise<void> {
    if (this.conn) return;
    if (this.initPromise) return this.initPromise;
    this.initPromise = (async () => {
      const backend = detectEncryptedBackend(this.cipherKey);
      this.conn = await backend.open(this.dbPath, { readonly: false, cipherKey: this.cipherKey });
      this.conn.exec(`...same CREATE TABLE...`);
    })();
    return this.initPromise;
  }

  // rest of methods unchanged
}
```
